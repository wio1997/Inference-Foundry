208:     def parallel_mode(self) -> SharedExpertParallelMode:
209:         """Resolve the effective activation/weight layout for this forward."""
210:         # EP rewrites FusedMoEParallelConfig.tp_size to 1 because each routed

212:         # group, so their layout must be derived from that group instead.

214:         if tp_size <= 1:

216:

218:             # SP has already sharded the token dimension before entering the
219:             # runner. Replicated weights (SP+DP) compute directly on the
220:             # shard; TP-sharded weights (SP-only) gather the shard first and
